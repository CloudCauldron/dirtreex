% The icon is measured and centred by dirtreex. This hook supplies only the
% glyph presentation: a borderless white circle with a coloured glyph.
\renewcommand{\DirtreexFormatIcon}[1]{%
  \tikz[baseline=0pt] \node[
    circle, fill=white, draw=none, inner sep=2pt
  ] {\textcolor{blue}{\DirtreexGetField{#1}{i}}};%
}
